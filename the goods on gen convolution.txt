// --- Initial setup ---
N = dim(TE_in, 2); // Get the number of channels
L = dim(H_kernel);  // Get the kernel length (should be N for full convolution)

// --- Loop through each output channel (i) ---
// You'll need a way to output N channels. The `out` operator
// will be used inside the loop, using the @channel attribute.

for (i = 0; i < N; i += 1) {
    // --- Calculate the value for the i-th output channel (y_i) ---
    sum = 0.0;

    // --- Loop through the input channels (k) and the kernel (h) ---
    // This is where you perform the dot product (the convolution sum)
    for (k = 0; k < N; k += 1) {
        // Calculate the kernel index: 
        // h_index = i - k. We use L - (k-i) to flip and align the kernel.
        // A simple "center-aligned" kernel for sharpening would be h[k]

        // 1. Read the input TE value from channel k
        // TE_in is a 1-sample, N-channel buffer
        x_k = peek(TE_in, 0, k); // Read frame 0, channel k

        // 2. Determine the corresponding kernel value h_j
        // For a center-aligned sharpening kernel, use an index relative to the center.
        // Let's assume a symmetric kernel loaded in H_kernel.
        h_j = peek(H_kernel, abs(k - i)); // e.g. difference from center channel

        sum += x_k * h_j;
    }

    // --- Output the result for channel i ---
    out(sum, @channel i); 
}
